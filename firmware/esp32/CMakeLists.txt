cmake_minimum_required(VERSION 3.16)
project(forklift_esp32_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(firmware_core forklift_esp32/firmware_core.cpp)
target_include_directories(firmware_core PUBLIC forklift_esp32)
target_compile_options(firmware_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(FIRMWARE_SANITIZE "Enable address/undefined behavior sanitizers on native tests" OFF)
if(FIRMWARE_SANITIZE)
  target_compile_options(firmware_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(firmware_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/serial_control_tests.cpp")
  add_executable(serial_control_tests tests/serial_control_tests.cpp)
  target_link_libraries(serial_control_tests PRIVATE firmware_core)
  target_compile_options(serial_control_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME serial_control_tests COMMAND serial_control_tests)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core_tests.cpp")
  find_package(GTest REQUIRED)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE firmware_core GTest::gtest_main)
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  include(GoogleTest)
  gtest_discover_tests(core_tests)
endif()
