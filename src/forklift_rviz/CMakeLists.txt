cmake_minimum_required(VERSION 3.8)
project(forklift_rviz)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(mission_plugins SHARED src/mission_plugins.cpp)
target_compile_features(mission_plugins PUBLIC cxx_std_17)
ament_target_dependencies(mission_plugins rviz_common rviz_default_plugins pluginlib rclcpp std_srvs std_msgs)
target_link_libraries(mission_plugins Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS mission_plugins LIBRARY DESTINATION lib)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/ui_smoke.cpp")
  add_executable(mission_ui_smoke test/ui_smoke.cpp)
  ament_target_dependencies(mission_ui_smoke rviz_common rclcpp)
  target_link_libraries(mission_ui_smoke Qt5::Widgets)
endif()

ament_package()
