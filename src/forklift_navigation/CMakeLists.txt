cmake_minimum_required(VERSION 3.8)
project(forklift_navigation)
find_package(ament_cmake REQUIRED)
find_package(dwb_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(forklift_goal_critic SHARED src/exact_goal_critic.cpp)
target_include_directories(forklift_goal_critic PUBLIC include)
ament_target_dependencies(forklift_goal_critic dwb_core pluginlib)
pluginlib_export_plugin_description_file(dwb_core critics.xml)
install(TARGETS forklift_goal_critic LIBRARY DESTINATION lib)
add_library(forklift_terminal_planner SHARED src/terminal_approach_planner.cpp)
target_include_directories(forklift_terminal_planner PUBLIC include)
ament_target_dependencies(forklift_terminal_planner nav2_smac_planner nav2_core nav2_costmap_2d nav2_util tf2_geometry_msgs nav_msgs geometry_msgs pluginlib)
pluginlib_export_plugin_description_file(nav2_core planners.xml)
install(TARGETS forklift_terminal_planner LIBRARY DESTINATION lib)
add_library(forklift_stable_controller SHARED src/stable_reversing_controller.cpp)
target_include_directories(forklift_stable_controller PUBLIC include)
ament_target_dependencies(forklift_stable_controller nav2_regulated_pure_pursuit_controller nav2_core nav2_costmap_2d tf2_geometry_msgs nav_msgs geometry_msgs pluginlib)
pluginlib_export_plugin_description_file(nav2_core controllers.xml)
install(TARGETS forklift_stable_controller LIBRARY DESTINATION lib)
add_library(forklift_mppi_controller SHARED src/guarded_mppi_controller.cpp)
target_include_directories(forklift_mppi_controller PUBLIC include)
ament_target_dependencies(forklift_mppi_controller nav2_mppi_controller nav2_core nav2_costmap_2d nav2_util tf2_geometry_msgs nav_msgs geometry_msgs pluginlib)
pluginlib_export_plugin_description_file(nav2_core mppi_controllers.xml)
install(TARGETS forklift_mppi_controller LIBRARY DESTINATION lib)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test")
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(terminal_path_test test/terminal_path_test.cpp)
  target_include_directories(terminal_path_test PRIVATE include)
  ament_target_dependencies(terminal_path_test nav2_costmap_2d tf2_geometry_msgs nav_msgs)
  ament_add_gtest(exact_goal_test test/exact_goal_test.cpp)
  ament_add_gtest(route_guard_test test/route_guard_test.cpp)
  target_include_directories(route_guard_test PRIVATE include)
  ament_target_dependencies(route_guard_test nav2_costmap_2d tf2_geometry_msgs nav_msgs)
  target_link_libraries(exact_goal_test forklift_goal_critic)
  ament_target_dependencies(exact_goal_test dwb_core)
endif()
install(DIRECTORY config launch maps DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/navigation_guard.py scripts/scan_relay.py DESTINATION lib/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
